# === Helper Constants ===
INVALID_FLOAT_VALUES = frozenset([None, "", "nan", "unknown", "Unknown"])
INVALID_DIMENSION_VALUES = frozenset([None, "", "unknown", "Unknown"])
INVALID_INT_VALUES = frozenset([None, "", "nan", "unknown", "Unknown"])
INVALID_STRING_VALUES = frozenset([None])  # Only None is truly invalid for strings

# === Helper Functions ===


def safe_float(value):
    """Safely converts a value to float, returning None for invalid inputs."""
    if isinstance(value, str):
        value = value.strip()
    if value in INVALID_FLOAT_VALUES:
        return None
    try:
        return float(value)
    except (ValueError, TypeError):
        return None


def safe_int(value):
    """
    Safely attempts to convert a value to an integer.
    Handles None, empty strings, 'nan', 'unknown', and float strings like '5.0'.
    Returns None if conversion is not possible or value is invalid.
    """
    if isinstance(value, str):
        value = value.strip()
    if value in INVALID_INT_VALUES:
        return None
    try:
        return int(float(value))
    except (ValueError, TypeError):
        return None


def safe_dimension(value):
    """Returns the value if it's a valid dimension string, else None."""
    if isinstance(value, str):
        value = value.strip()
        if value.lower() in INVALID_DIMENSION_VALUES:
            return None
    elif value in INVALID_DIMENSION_VALUES:
        return None
    return value


def safe_string(value):
    """Returns the stripped string value, or None if the input is None or results in empty string."""
    if isinstance(value, str):
        stripped = value.strip()
        return stripped if stripped else None  # Treat empty string as None
    return value if value not in INVALID_STRING_VALUES else None

def get_redis_connection():
    try:
        # Basic redis-py connection pool
        pool = redis.ConnectionPool(
            host=settings.REDIS_HOST,
            port=settings.REDIS_PORT,
            db=settings.REDIS_DB_TEMP_FLOORPLANS,
            decode_responses=True,  # Decode responses to strings
        )
        return redis.Redis(connection_pool=pool)
        # OR if using django-redis cache backend named 'temp_floorplans':
        # from django.core.cache import caches
        # return caches['temp_floorplans'].client.get_client()
    except Exception as e:
        logger.error(f"Failed to connect to Redis for temp floorplan storage: {e}")
        return None


@shared_task(bind=True)
def process_floorplan_analysis(self, user_id, property_id, floorplans):
    """
    Task to initiate the analysis process via the analyzer API.
    """
    logger.info(
        "Starting floorplan analysis task for user_id=%s, property_id=%s",
        user_id,
        property_id,
    )
    # webhook_url = settings.FLOORPLAN_WEBHOOK_URL
    webhook_url = "https://9687-41-76-192-17.ngrok-free.app/api/floorplan/webhook/"
    analyzer_url = settings.FLOORPLAN_ANALYZER_URL

    if not webhook_url or not analyzer_url:
        logger.error(
            "Missing FLOORPLAN_WEBHOOK_URL or FLOORPLAN_ANALYZER_URL in settings."
        )
        return {"error": "Server configuration error: Missing required URLs."}

    # --- Store original data in Redis ---
    redis_conn = get_redis_connection()
    redis_key = f"original_floorplans:{user_id}:{property_id}"
    stored_successfully = False
    if redis_conn:
        try:
            # Store the floorplans dict as a JSON string
            redis_conn.set(
                redis_key, json.dumps(floorplans), ex=settings.REDIS_TEMP_FLOORPLANS_TTL
            )
            logger.info(
                "Successfully stored original floorplan data in Redis for key: %s",
                redis_key,
            )
            stored_successfully = True
        except Exception as e:
            logger.error(
                "Failed to store original floorplan data in Redis for key %s: %s",
                redis_key,
                e,
            )
            # Decide if failure to store should prevent calling the API
            # For now, we'll log the error and continue, but the fallback won't work.
    else:
        logger.error(
            "Could not get Redis connection. Fallback URL logic will not work."
        )
    # --- End Redis storage ---

    payload = {
        "webhook_url": webhook_url,
        "user_id": user_id,
        "property_id": property_id,
        "floorplans": floorplans,
    }

    try:
        logger.info(
            "Calling analyzer API at %s with payload: %s", analyzer_url, payload
        )
        response = requests.post(analyzer_url, json=payload, timeout=60)  # Add timeout
        response.raise_for_status()
        logger.info("Analyzer API call successful, analysis initiated.")
    except requests.Timeout:
        logger.error("Timeout calling analyzer API at %s", analyzer_url)
        raise self.retry(countdown=60, max_retries=5)
    except requests.RequestException as e:
        status_code = getattr(e.response, "status_code", "N/A")
        response_text = getattr(e.response, "text", "N/A")
        logger.error(
            "Failed to call analyzer API: %s (Status: %s). Response: %s",
            str(e),
            status_code,
            response_text,
        )
        if (
            redis_conn
            and stored_successfully
            and status_code not in [500, 502, 503, 504]
        ):  # Don't delete on potentially temporary server errors
            try:
                redis_conn.delete(redis_key)
                logger.info(
                    "Deleted temporary Redis key %s due to definitive API call failure.",
                    redis_key,
                )
            except Exception as del_e:
                logger.error(
                    "Failed to delete Redis key %s after API failure: %s",
                    redis_key,
                    del_e,
                )
        raise self.retry(exc=e, countdown=60, max_retries=5)
    except Exception as e:
        logger.exception(
            "Unexpected error during analyzer API call preparation/execution: %s", e
        )
        raise self.retry(exc=e, countdown=120, max_retries=3)

    logger.info(
        "Floorplan analysis initiation task completed for user_id=%s, property_id=%s",
        user_id,
        property_id,
    )
    return {"message": "Analysis initiated successfully"}


@shared_task(bind=True)
@transaction.atomic
def process_floorplan_webhook(self, analysis_data):
    """
    Processes the analysis payload sent to the webhook (entry point).
    Delegates processing based on the task type ('creation' or 'update').
    """
    task_type = analysis_data.get("task")
    user_id = analysis_data.get("user_id")
    property_id = analysis_data.get("property_id")
    output_data = analysis_data.get("output_data")  # Extract output_data

    # --- Retrieve original data from Redis ---
    original_request_floorplans = None
    redis_conn = get_redis_connection()
    redis_key = f"original_floorplans:{user_id}:{property_id}"
    if redis_conn:
        try:
            stored_data = redis_conn.get(redis_key)
            if stored_data:
                original_request_floorplans = json.loads(stored_data)  # Deserialize
                logger.info(
                    "Successfully retrieved original floorplan data from Redis for key: %s",
                    redis_key,
                )
                # Optional: Delete key after retrieval if it's single-use
                # redis_conn.delete(redis_key)
            else:
                logger.warning(
                    "No original floorplan data found in Redis for key: %s (expired or not stored?)",
                    redis_key,
                )
        except json.JSONDecodeError:
            logger.error("Failed to decode JSON data from Redis key: %s", redis_key)
        except Exception as e:
            logger.error(
                "Failed to retrieve/decode data from Redis for key %s: %s", redis_key, e
            )
    else:
        logger.warning(
            "Could not get Redis connection. Fallback URL logic will not work."
        )
    # --- End Redis retrieval ---

    # 1. Check base required fields (excluding output_data for now)
    base_required = {"task": task_type, "user_id": user_id, "property_id": property_id}
    missing_base = [k for k, v in base_required.items() if not v]
    if missing_base:
        logger.error(
            "Webhook received incomplete base data. Missing: %s. Payload: %s",
            missing_base,
            analysis_data,
        )
        return {
            "error": f"Missing required base fields in webhook payload: {missing_base}"
        }

    # 2. Check if output_data exists and is a list (handle non-list types)
    if output_data is None:
        missing_fields = ["output_data"]  # Explicitly state it's missing
        logger.error(
            "Webhook received incomplete data. Missing: %s. Payload: %s",
            missing_fields,
            analysis_data,
        )
        return {
            "error": f"Missing required fields in webhook payload: {missing_fields}"
        }

    if not isinstance(output_data, list):
        logger.error(
            "Webhook received invalid type for output_data: %s. Expected list. Payload: %s",
            type(output_data).__name__,
            analysis_data,
        )
        return {
            "error": f"Invalid output_data type in webhook payload: expected list, got {type(output_data).__name__}"
        }

    # 3. Handle the case where output_data is an EMPTY list
    if not output_data:  # Specifically checks if the list is empty
        logger.warning(
            "Webhook received empty output_data list for user %s, property %s. Assuming no floorplans were processed or returned by extractor. Payload: %s",
            user_id,
            property_id,
            analysis_data,
        )
        # Optionally: Still record that the analysis attempt happened
        try:
            analysis_result, created = FloorPlanAnalysisResult.objects.update_or_create(
                user_id=user_id,
                property_id=property_id,
                defaults={
                    "message": analysis_data.get(
                        "message", "Extractor returned no floorplan data"
                    )
                },
            )
            action = "Created" if created else "Updated"
            logger.info(
                "%s FloorPlanAnalysisResult (ID: %s) for user %s, property %s, received empty output_data.",
                action,
                analysis_result.id,
                user_id,
                property_id,
            )
        except Exception as db_exc:
            logger.error(
                "Error updating FloorPlanAnalysisResult for empty webhook data (user %s, property %s): %s",
                user_id,
                property_id,
                db_exc,
            )
        if redis_conn and stored_data:  # Check if we had data to delete
            try:
                redis_conn.delete(redis_key)
                logger.info(
                    "Deleted Redis key %s after processing empty webhook.", redis_key
                )
            except Exception as del_e:
                logger.error(
                    "Failed to delete Redis key %s after empty webhook: %s",
                    redis_key,
                    del_e,
                )
        return {
            "message": "Webhook processed successfully, but output_data list was empty."
        }

    logger.info(
        "Processing webhook task '%s' for user '%s', property '%s' with %d item(s) in output_data",
        task_type,
        user_id,
        property_id,
        len(output_data),  # Now safe to use len()
    )

    if not original_request_floorplans:
        logger.warning(
            "Proceeding without original request data from Redis. Fallback URL logic disabled."
        )

    try:
        if task_type == "creation":
            result = _handle_creation_task(analysis_data, original_request_floorplans)
        elif task_type == "update":
            # NOTE: With hash IDs, 'update' task type from analyzer might be less relevant,
            # as we identify floorplans by URL hash. Handling both for compatibility.
            result = _handle_update_task(analysis_data, original_request_floorplans)
        else:
            logger.error("Unknown task type received in webhook: %s", task_type)
            result = {"error": f"Unknown task type: {task_type}"}

        # --- Optional: Delete Redis key after successful processing ---
        if (
            redis_conn
            and original_request_floorplans
            and result
            and not result.get("error")
        ):  # Check if processing likely succeeded
            try:
                redis_conn.delete(redis_key)
                logger.info(
                    "Deleted Redis key %s after successful webhook processing.",
                    redis_key,
                )
            except Exception as del_e:
                logger.error(
                    "Failed to delete Redis key %s after successful processing: %s",
                    redis_key,
                    del_e,
                )
        # --- End optional deletion ---

        logger.info(
            "Webhook processing completed for task '%s', user '%s', property '%s'. Result: %s",
            task_type,
            user_id,
            property_id,
            result,
        )
        return result

    except Exception as e:
        logger.exception(
            "Unhandled error during webhook processing for task '%s', user '%s', property '%s': %s",
            task_type,
            user_id,
            property_id,
            str(e),
        )
        return {"error": f"Internal server error during webhook processing: {str(e)}"}


# --- Private Helper Functions for Webhook Processing ---


def _handle_creation_task(analysis_data, original_request_floorplans=None):
    """Handles the 'creation' task logic."""
    user_id = analysis_data["user_id"]
    property_id = analysis_data["property_id"]
    output_data = analysis_data[
        "output_data"
    ]  # This is expected to be the LIST of floorplan results
    message = analysis_data.get("message", "Floorplan batch created")

    if not isinstance(output_data, list):
        logger.error(
            "Webhook 'creation' task expected list for output_data, got %s",
            type(output_data),
        )
        return {"error": "Invalid output_data format for creation task."}

    analysis_result, created = FloorPlanAnalysisResult.objects.update_or_create(
        user_id=user_id,
        property_id=property_id,
        defaults={"message": message},  # Update message on each creation/update
    )
    action = "Created" if created else "Updated"
    logger.info(
        "%s FloorPlanAnalysisResult (ID: %s) for user %s, property %s.",
        action,
        analysis_result.id,
        user_id,
        property_id,
    )

    processed_stable_ids = set()
    # Process each floorplan item in the list
    for item_data in output_data:
        # Assume item_data is a dictionary representing one floorplan's analysis result
        # It should contain keys like 'floorplan_id' (original request key), 'original_url', 'all_floors', 'floors'
        api_returned_key = item_data.get("floorplan_id")  # e.g. "fp1"
        if not api_returned_key:
            if not api_returned_key:
                logger.warning(
                    "Skipping item in creation list due to missing or potentially invalid 'floorplan_id' key from API: %s",
                    item_data.get("floorplan_id", "N/A"),
                )
            continue
        try:
            # Process this single floorplan's data, generating a stable ID inside
            processed_fp_instance = _process_single_floorplan_data(
                item_data, analysis_result, original_request_floorplans
            )
            if processed_fp_instance:
                processed_stable_ids.add(
                    processed_fp_instance.floorplan_id
                )  # Add the stable ID (hash)
        except Exception as e:
            logger.exception(
                "Error processing floorplan item (API Key: %s) during creation for AnalysisResult %s: %s",
                api_returned_key,
                analysis_result.id,
                str(e),
            )
            # Continue processing other items in the batch

    logger.info(
        "Finished processing 'creation' task for AnalysisResult %s. Processed floorplan stable IDs: %s",
        analysis_result.id,
        processed_stable_ids,
    )
    return {
        "message": "Webhook creation processing completed",
        "analysis_id": analysis_result.id,
        "processed_floorplan_ids": list(
            processed_stable_ids
        ),  # Return the stable IDs processed
    }


def _handle_update_task(analysis_data):
    """
    Handles the 'update' task logic.
    NOTE: With hash IDs, this might be less common. It processes a single floorplan update.
    The payload 'output_data' is expected to be a DICT for one floorplan.
    """
    user_id = analysis_data["user_id"]
    property_id = analysis_data["property_id"]
    output_data = analysis_data[
        "output_data"
    ]  # Expected to be a DICT for one floorplan result

    if not isinstance(output_data, dict):
        logger.error(
            "Webhook 'update' task expected dict for output_data, got %s",
            type(output_data),
        )
        return {"error": "Invalid output_data format for update task."}

    # The 'floorplan_id' in the update payload is likely the original request key ('fp1')
    # We need the original_url to find/update the correct record via hash.
    original_request_key = output_data.get("floorplan_id")
    original_url = output_data.get("original_url")

    if not original_request_key or not original_url:
        logger.error(
            "Missing floorplan_id (original key) or original_url in update payload: %s",
            output_data,
        )
        return {"error": "Missing floorplan_id or original_url in update payload"}

    try:
        # Find the *AnalysisResult* first, as FloorPlan ID is now derived
        analysis_result = FloorPlanAnalysisResult.objects.get(
            user_id=user_id, property_id=property_id
        )

        logger.info(
            "Processing update for AnalysisResult ID %s (User: %s, Prop: %s)",
            analysis_result.id,
            user_id,
            property_id,
        )

        # Use the hash of the URL to find/update the FloorPlan
        stable_floorplan_id = FloorPlan.generate_hash_id(original_url)
        logger.info(
            "Processing update for AnalysisResult ID %s, attempting to find/update FloorPlan with Hash ID %s",
            analysis_result.id,
            stable_floorplan_id,
        )

        # Process this single floorplan's data, generating/using the stable ID inside
        processed_fp_instance = _process_single_floorplan_data(
            output_data, analysis_result, is_update=True
        )  # Pass is_update flag if needed

        if not processed_fp_instance:
            return {
                "error": f"Failed to process update for floorplan URL: {original_url}"
            }

        logger.info(
            "Finished processing 'update' task for floorplan stable ID %s.",
            stable_floorplan_id,
        )

        return {
            "message": "Floor update processing completed",
            "floorplan_id": stable_floorplan_id,
            "analysis_id": analysis_result.id,
        }

    except FloorPlanAnalysisResult.DoesNotExist:
        logger.error(
            "FloorPlanAnalysisResult for user '%s', property '%s' does not exist for update.",
            user_id,
            property_id,
        )
        return {
            "error": f"FloorPlanAnalysisResult for {user_id}/{property_id} not found for update."
        }
    except ValueError as ve:  # Catch hashing errors
        logger.error("Error generating hash during update task: %s", ve)
        return {"error": f"Failed to generate hash ID for update: {ve}"}
    except Exception as e:
        logger.exception(
            "Error during update task for floorplan (Original Key: '%s'): %s",
            original_request_key,
            str(e),
        )
        return {
            "error": f"Failed to process update for floorplan (Original Key: {original_request_key})."
        }


def _process_single_floorplan_data(
    item_data, analysis_result, original_request_floorplans=None, is_update=False
):
    """
    Processes the data for a single floorplan item received from the webhook.
    Generates a stable ID hash from the original_url.
    Uses update_or_create for FloorPlan based on hash ID.
    Returns the processed FloorPlan instance or None on failure.
    """
    api_returned_key = item_data.get("floorplan_id")  # e.g., "fp1"
    api_returned_url = item_data.get("original_url")
    all_floors_item = item_data.get("all_floors")  # Dict for AllFloorsData URLs etc.
    floors_list = item_data.get("floors", [])  # List for PlanFloor data

    if not all_floors_item:
        logger.warning(
            "Skipping floorplan item processing (API Key: %s) due to missing 'all_floors' data in webhook item: %s",
            api_returned_key,
            item_data,
        )  # Indicate failure

    url_to_hash = None
    source_url_for_db = None

    if (
        api_returned_url
        and isinstance(api_returned_url, str)
        and api_returned_url.strip()
    ):
        url_to_hash = api_returned_url.strip()
        source_url_for_db = url_to_hash
        logger.info("Using original_url directly from API response: %s", url_to_hash)
    elif original_request_floorplans and api_returned_key:
        logger.warning(
            "API did not return a valid original_url for key '%s'. Attempting fallback using original request data from Redis.",
            api_returned_key,
        )
        original_request_item = original_request_floorplans.get(api_returned_key)
        if original_request_item and isinstance(original_request_item, dict):
            fallback_url = original_request_item.get("url")
            if fallback_url and isinstance(fallback_url, str) and fallback_url.strip():
                url_to_hash = fallback_url.strip()
                source_url_for_db = url_to_hash
                logger.warning(
                    "Fallback successful: Using original request URL '%s' for API key '%s'.",
                    url_to_hash,
                    api_returned_key,
                )
            else:
                logger.error(
                    "Fallback failed: Found original request item for key '%s', but its URL is missing or invalid.",
                    api_returned_key,
                )
        else:
            # Handle if the key from API doesn't match keys in stored data
            logger.error(
                "Fallback failed: Could not find original request data in Redis for the key '%s' returned by the API.",
                api_returned_key,
            )
    else:
        logger.error(
            "Cannot determine URL for hashing. API Key: '%s'. API URL: '%s'. Original request data available: %s",
            api_returned_key,
            api_returned_url,
            bool(original_request_floorplans),
        )

    if not url_to_hash:
        logger.error(
            "Failed to obtain a valid URL for floorplan item. Skipping processing. Item data: %s",
            item_data,
        )
        return None

    # --- Generate ID based on URL Hash using Model's static method ---
    try:
        stable_floorplan_id = FloorPlan.generate_hash_id(url_to_hash)
    except ValueError as hash_exc:  # Catch specific error from generate_hash_id
        logger.error(
            f"Failed to hash URL '{url_to_hash}'. Skipping this floorplan. Error: {hash_exc}"
        )
        return None  # Indicate failure

    logger.info(
        f"Processing floorplan item (API Key: {api_returned_key}, URL Used for Hash: {url_to_hash}, Assigned Stable ID (Hash): {stable_floorplan_id})"
    )

    # 1. Update or Create FloorPlan using the STABLE HASH ID
    #    Use floorplan_id=stable_floorplan_id and ensure it's unique=True in model
    floorplan_defaults = {
        "analysis_result": analysis_result,  # Ensure analysis_result is passed in defaults for creation
        "original_url": source_url_for_db,
    }
    try:
        floorplan, fp_created = FloorPlan.objects.update_or_create(
            floorplan_id=stable_floorplan_id, defaults=floorplan_defaults
        )
    except Exception as db_exc:
        logger.exception(
            "Database error during FloorPlan update_or_create for hash %s (URL: %s): %s",
            stable_floorplan_id,
            source_url_for_db,
            db_exc,
        )
        return None
    action_fp = "Created" if fp_created else "Updated"
    logger.info("%s FloorPlan with Stable ID (Hash)=%s", action_fp, stable_floorplan_id)

    # 2. Update or Create AllFloorsData
    all_floors_defaults = {
        "json_file_url": all_floors_item.get("json_file_url"),
        "csv_url": all_floors_item.get("csv_url"),
        "total_area_csv_url": all_floors_item.get("total_area_csv_url"),
        "image_labelme_side_by_side_url": all_floors_item.get(
            "image_labelme_side_by_side_url"
        ),
        "notes": all_floors_item.get("notes", ""),
    }
    all_floors_defaults = {
        k: v for k, v in all_floors_defaults.items() if v is not None
    }

    all_floors_data, afd_created = AllFloorsData.objects.update_or_create(
        floor_plan=floorplan,  # Link using the FloorPlan object
        defaults=all_floors_defaults,
    )
    action_afd = "Created" if afd_created else "Updated"
    logger.info(
        "%s AllFloorsData for Stable Floorplan ID=%s", action_afd, stable_floorplan_id
    )

    # 3. Update or Create PlanFloors (handle deletions)
    _update_plan_floors(floorplan, floors_list)

    # 4. Process the main CSV (all_floors.csv) -> Structured data + Raw Rows
    _process_main_csv(all_floors_data)

    # 5. Process the total_area.csv -> TotalAreaData model
    _process_total_area_csv(all_floors_data)

    # 6. Backup the processed state
    _perform_backup(floorplan)

    return floorplan  # Return the processed instance


def _update_plan_floors(floorplan, floors_list_data):
    """Updates PlanFloor entries based on the provided list, deleting orphans."""
    existing_plan_floor_names = set(
        floorplan.plan_floors.values_list("floor", flat=True)
    )
    current_plan_floor_names = set()

    for floor_data in floors_list_data:
        floor_name = floor_data.get("floor")
        if not floor_name:
            logger.warning(
                "Skipping plan floor item due to missing floor name for floorplan %s: %s",
                floorplan.floorplan_id,
                floor_data,
            )
            continue
        current_plan_floor_names.add(floor_name)

        plan_floor_defaults = {
            "label_me_url": floor_data.get("label_me_url"),
            "json_file_url": floor_data.get("json_file_url"),
            "image_url": floor_data.get("image_url"),
            "labelme_image_url": floor_data.get("labelme_image_url"),
            "csv_url": floor_data.get("csv_url"),
            "image_side_by_side_url": floor_data.get("image_side_by_side_url"),
        }
        plan_floor_defaults = {
            k: v for k, v in plan_floor_defaults.items() if v is not None
        }

        pf, pf_created = PlanFloor.objects.update_or_create(
            floor_plan=floorplan,
            floor=floor_name,
            defaults=plan_floor_defaults,
        )
        # action_pf = "Created" if pf_created else "Updated" # Less verbose logging

    floors_to_remove = existing_plan_floor_names - current_plan_floor_names
    if floors_to_remove:
        logger.info(
            "Removing %d old PlanFloors for floorplan %s not in new data: %s",
            len(floors_to_remove),
            floorplan.floorplan_id,
            floors_to_remove,
        )
        floorplan.plan_floors.filter(floor__in=floors_to_remove).delete()


def _download_csv_content(csv_url, context_msg):
    """Downloads CSV content from a URL, handles errors."""
    if not csv_url:
        logger.warning("No CSV URL provided for %s.", context_msg)
        return None
    try:
        logger.info("Downloading CSV for %s from %s", context_msg, csv_url)
        response = requests.get(csv_url, timeout=30)
        response.raise_for_status()
        return response.text
    except requests.Timeout:
        logger.error("Timeout downloading CSV for %s from %s", context_msg, csv_url)
        return None
    except requests.RequestException as e:
        logger.error(
            "Failed to download CSV for %s from %s: %s", context_msg, csv_url, str(e)
        )
        return None


def _process_main_csv(all_floors_data):

def process_structured_csv_data(rows, all_floors_data):
    """
    Processes CSV rows into structured models (CsvFloor, CsvRoom, etc.).
    Uses update_or_create and handles deletion of orphans.
    Uses 'unspecified_floor' for blank/unknown floor names.
    """
    processed_floor_names = set()
    processed_room_ids_by_floor = {}
    floor_cache = {}

    for row in rows:
        floor_name_raw = row.get("Floor_Name")
        floor_name = safe_string(floor_name_raw)  # Handles None, trims whitespace

        # Assign a default if the name is missing, empty, or explicitly 'unknown'
        if not floor_name or floor_name.lower() == "unknown":
            floor_name = "unspecified_floor"  # Single placeholder

        room_id_str = row.get("Room_id")
        room_id = safe_float(room_id_str)

        if room_id is None:
            logger.warning(
                "Skipping CSV row for floor '%s' due to invalid/missing Room_id ('%s'). Row snippet: %s",
                floor_name,
                room_id_str,
                {
                    k: v
                    for k, v in row.items()
                    if k in ["Room_id", "Room_Name", "Floor_Name"]
                },
            )
            continue

        processed_floor_names.add(floor_name)
        processed_room_ids_by_floor.setdefault(floor_name, set()).add(room_id)

        # --- Process CsvFloor ---
        if floor_name not in floor_cache:
            floor, f_created = CsvFloor.objects.update_or_create(
                all_floors_data=all_floors_data,
                floor_name=floor_name,
                defaults={
                    "calculated_total_area_metric": safe_float(
                        row.get("Calculated Floor Total Sq Area Metric")
                    ),
                    "calculated_total_area_imperial": safe_float(
                        row.get("Calculated Floor Total Sq Area Imperial")
                    ),
                },
            )
            floor_cache[floor_name] = floor
        else:
            floor = floor_cache[floor_name]

        # --- Process CsvRoom ---
        room_defaults = {
            "room_name": safe_string(row.get("Room_Name")),
            "is_segment": safe_string(row.get("is_segment")),
            "no_of_doors": safe_float(row.get("No_of_door")),
            "no_of_windows": safe_float(row.get("No_of_window")),
            "no_of_room_points": safe_float(row.get("No_of_room_points")),
        }
        room, r_created = CsvRoom.objects.update_or_create(
            floor=floor,
            room_id=room_id,
            defaults=room_defaults,
        )

        _update_or_create_room_details(room, row)

    # --- Clean up old structured data ---
    floors_to_delete = all_floors_data.csv_floors.exclude(
        floor_name__in=processed_floor_names
    )
    if floors_to_delete.exists():
        deleted_floor_names = list(
            floors_to_delete.values_list("floor_name", flat=True)
        )
        logger.info(
            "Deleting %d CsvFloors (and related rooms) for AllFloorsData %d no longer present in CSV: %s",
            floors_to_delete.count(),
            all_floors_data.id,
            deleted_floor_names,
        )
        floors_to_delete.delete()

    for floor_name, processed_room_ids in processed_room_ids_by_floor.items():
        if floor_name in floor_cache:
            csv_floor = floor_cache[floor_name]
            rooms_to_delete = csv_floor.rooms.exclude(room_id__in=processed_room_ids)
            if rooms_to_delete.exists():
                deleted_room_ids = list(
                    rooms_to_delete.values_list("room_id", flat=True)
                )
                logger.info(
                    "Deleting %d CsvRooms for floor '%s' (AllFloorsData %d) no longer present in CSV: %s",
                    rooms_to_delete.count(),
                    floor_name,
                    all_floors_data.id,
                    deleted_room_ids,
                )
                rooms_to_delete.delete()

    logger.info(
        "Structured CSV data processing finished for AllFloorsData id=%s",
        all_floors_data.id,
    )


def _update_or_create_room_details(room, row_data):
    """Helper to update/create related OneToOne room details."""
    # --- Pixel Data ---
    pixel_defaults = {
        "min_x_pixels": safe_float(row_data.get("Min X Pixels")),
        "min_y_pixels": safe_float(row_data.get("Min Y Pixels")),
        "max_x_pixels": safe_float(row_data.get("Max X Pixels")),
        "max_y_pixels": safe_float(row_data.get("Max Y Pixels")),
        "max_area_pixels": safe_float(row_data.get("Max Area Pixels")),
        "actual_area_pixels": safe_float(row_data.get("Actual Area Pixels")),
        "pixel_ratio": safe_float(row_data.get("Pixel ratio")),
    }
    if any(v is not None for v in pixel_defaults.values()):
        CsvRoomPixelData.objects.update_or_create(room=room, defaults=pixel_defaults)
    else:
        CsvRoomPixelData.objects.filter(room=room).delete()

    # --- Dimensions ---
    dimension_defaults = {
        "dimensions_imperial": safe_dimension(row_data.get("dimensions_imperial")),
        "dimensions_metric": safe_dimension(row_data.get("dimensions_metric")),
        "max_area_metric": safe_float(row_data.get("Max Area Metric")),
        "max_area_imperial": safe_float(row_data.get("Max Area Imperial")),
        "calculated_sq_area_metric": safe_float(
            row_data.get("Calculated Sq Area Metric")
        ),
        "calculated_area_imperial": safe_float(
            row_data.get("calculated_area_imperial")
        ),
    }
    if any(v is not None for v in dimension_defaults.values()):
        CsvRoomDimensions.objects.update_or_create(
            room=room, defaults=dimension_defaults
        )
    else:
        CsvRoomDimensions.objects.filter(room=room).delete()

    # --- Scaling Factors ---
    scaling_defaults = {
        "scale_metric": safe_float(row_data.get("Scale Metric")),
        "scale_imperial": safe_float(row_data.get("Scale Imperial")),
    }
    if any(v is not None for v in scaling_defaults.values()):
        CsvRoomScalingFactors.objects.update_or_create(
            room=room, defaults=scaling_defaults
        )
    else:
        CsvRoomScalingFactors.objects.filter(room=room).delete()

from rest_framework import status
from rest_framework.response import Response
from rest_framework.views import APIView

from floorplan.tasks import process_floorplan_analysis, process_floorplan_webhook


class FloorPlanAnalysisView(APIView):
    """
    Endpoint to initiate floorplan analysis.
    Expects a payload with "user_id", "property_id", and "floorplans".
    """

    def post(self, request):
        user_id = request.data.get("user_id")
        property_id = request.data.get("property_id")
        floorplans = request.data.get("floorplans")
        if not all([user_id, property_id, floorplans]):
            return Response(
                {"error": "Missing required fields: user_id, property_id, floorplans"},
                status=status.HTTP_400_BAD_REQUEST,
            )
        process_floorplan_analysis.delay(user_id, property_id, floorplans)
        return Response(
            {"message": "Floorplan analysis initiated."},
            status=status.HTTP_202_ACCEPTED,
        )


class FloorPlanWebhookView(APIView):
    """
    Webhook endpoint to receive analysis results from the analyzer service.

    Supports two kinds of payloads:
      - Creation: expects "task": "creation", "message", "user_id", "property_id",
        and "output_data" (a list of new floorplans).
      - Update: expects "task": "update", "user_id", "property_id",
        and "output_data" as an object containing a "floorplan_id" and a "floors" update.
    """

    def post(self, request):
        data = request.data
        required_fields = ["task", "user_id", "property_id", "output_data"]
        missing_fields = [field for field in required_fields if field not in data]
        if missing_fields:
            return Response(
                {"error": f"Missing fields in payload: {missing_fields}"},
                status=status.HTTP_400_BAD_REQUEST,
            )
        process_floorplan_webhook.delay(data)
        return Response(
            {"message": "Webhook received. Processing initiated."},
            status=status.HTTP_202_ACCEPTED,
        )

class Command(BaseCommand):
    help = "Exports floorplan data into three separate CSV files, including analysis creation time."

    def add_arguments(self, parser):
        parser.add_argument(
            "--output-dir",
            type=str,
            default=".",  # Default to current directory
            help="Directory where the CSV files will be saved.",
        )

    def handle(self, *args, **options):
        output_dir = options["output_dir"]
        os.makedirs(output_dir, exist_ok=True)  # Ensure directory exists

        csv1_path = os.path.join(output_dir, "floorplan_summary_export.csv")
        csv2_path = os.path.join(output_dir, "floorplan_room_details_export.csv")
        csv3_path = os.path.join(output_dir, "floorplan_total_area_export.csv")

        self.stdout.write(f"Starting export to directory: {output_dir}")

        try:
            self._export_csv1(csv1_path)
            self._export_csv2(csv2_path)
            self._export_csv3(csv3_path)
            self.stdout.write(self.style.SUCCESS("Successfully exported all data."))
        except Exception as e:
            self.stderr.write(self.style.ERROR(f"An error occurred during export: {e}"))
            traceback.print_exc()  # Print traceback for debugging
            raise CommandError(f"Export failed. See traceback above for details.")

    def _get_safe_value(self, obj, attribute_path, default=""):
        """Safely retrieves nested attributes, returning default if any part is None."""
        value = obj
        try:
            for attr in attribute_path.split("."):
                if value is None:
                    return default
                value = getattr(value, attr)
            return value if value is not None else default
        except AttributeError:
            return default

    def _format_datetime(self, dt_obj, default=""):
        """Formats a datetime object to ISO 8601 string, handling None."""
        if dt_obj and hasattr(dt_obj, "isoformat"):
            # Optional: Convert to local timezone if desired, otherwise uses stored timezone (likely UTC)
            # return timezone.localtime(dt_obj).isoformat()
            return dt_obj.isoformat()
        return default

    # === CSV 1 Export Logic ===
    def _export_csv1(self, filepath):
        self.stdout.write(f"Exporting CSV 1: Summary Data to {filepath}...")
        headers = [
            "property_id",
            "user_id",
            "floorplan_id",
            "original_url",
            "json_file_url",
            "csv_url",
            "total_area_csv_url",
            "image_labelme_side_by_side_url",
            "created_at",  # Changed header name
        ]

        queryset = AllFloorsData.objects.select_related(

import csv
import logging

from django.core.management.base import BaseCommand, CommandError

# Import relevant models
from floorplan.models import AllFloorsData, FloorPlan, FloorPlanAnalysisResult

logger = logging.getLogger(__name__)


class Command(BaseCommand):
    help = "Generates a CSV report mapping original properties to their final AllFloors and TotalArea CSV URLs."

    def add_arguments(self, parser):
        parser.add_argument(
            "input_csv_path",
            type=str,
            help="Path to the *original* input CSV file used for triggering analysis (to get the list of properties).",
        )
        parser.add_argument(
            "output_report_csv_path",
            type=str,
            help="Path to save the final report CSV file.",
        )
        # Optional: Add filters if you don't want to use the input CSV
        # parser.add_argument('--user-id', type=str, help='Filter report by User ID.')
        # parser.add_argument('--property-id', type=str, help='Filter report by Property ID.')

    def handle(self, *args, **options):
        input_csv_path = options["input_csv_path"]
        output_report_csv_path = options["output_report_csv_path"]

        report_data = []
        output_headers = [
            "Property ID",
            "User ID",
            "Property URL",
            "Original Floorplan URLs",
            "Processed Floorplan Hash IDs",  # Added
            "All Floors CSV URLs",
            "Total Area CSV URLs",
        ]
        report_data.append(output_headers)

        self.stdout.write(f"Reading property list from: {input_csv_path}")
        self.stdout.write(f"Generating final report at: {output_report_csv_path}")

        try:
            with open(input_csv_path, mode="r", encoding="utf-8-sig") as infile:
                reader = csv.DictReader(infile)
                if not all(h in reader.fieldnames for h in ["Property ID", "User ID"]):
                    raise CommandError(
                        "Input CSV must contain at least 'Property ID' and 'User ID' columns."
                    )

                processed_properties = (
                    set()
                )  # To avoid duplicate lookups if input has duplicates

                for i, row in enumerate(reader, start=2):
                    property_id_csv = row.get("Property ID", "").strip()
                    user_id_csv = row.get("User ID", "").strip()
                    property_url_csv = row.get(
                        "Property URL", ""
                    ).strip()  # Original property URL
                    original_fp_urls_csv = row.get(
                        "Floorplan URLs", ""
                    ).strip()  # Original list

                    if not property_id_csv or not user_id_csv:
                        self.stderr.write(
                            self.style.WARNING(
                                f"Skipping row {i} in input CSV: Missing Property ID or User ID."
                            )
                        )
                        continue

                    # Avoid reprocessing the same property_id/user_id combo
                    property_key = (user_id_csv, property_id_csv)
                    if property_key in processed_properties:
                        continue
                    processed_properties.add(property_key)

                    self.stdout.write(
                        f"Looking up results for User '{user_id_csv}', Property '{property_id_csv}'..."
                    )

                    all_floors_urls = []
                    total_area_urls = []
                    processed_fp_hashes = []

                    try:
                        # Find the analysis result
                        analysis_result = FloorPlanAnalysisResult.objects.get(
                            user_id=user_id_csv, property_id=property_id_csv
                        )

                        # Find all FloorPlan instances linked to this result
                        floorplans = (
                            FloorPlan.objects.filter(analysis_result=analysis_result)
                            .select_related("all_floors_data")
                            .order_by("id")
                        )  # Fetch related data efficiently

                        if not floorplans.exists():
                            self.stderr.write(
                                self.style.WARNING(
                                    f"  No FloorPlan records found for {property_key}."
                                )
                            )
                        else:
                            self.stdout.write(
                                f"  Found {floorplans.count()} processed floorplan(s)."
                            )
                            for fp in floorplans:
                                processed_fp_hashes.append(
                                    fp.floorplan_id
                                )  # Store the hash ID
                                try:
                                    # Access related AllFloorsData (already fetched via select_related)
                                    afd = fp.all_floors_data
                                    if afd:
                                        if afd.csv_url:
                                            all_floors_urls.append(afd.csv_url)
                                        if afd.total_area_csv_url:
                                            total_area_urls.append(
                                                afd.total_area_csv_url
                                            )
                                except AllFloorsData.DoesNotExist:
                                    self.stderr.write(
                                        self.style.WARNING(
                                            f"    AllFloorsData missing for FloorPlan Hash {fp.floorplan_id}."
                                        )
                                    )
                                except (
                                    AttributeError
                                ):  # Handle case where all_floors_data relation might not exist momentarily
                                    self.stderr.write(
                                        self.style.WARNING(
                                            f"    Error accessing all_floors_data for FloorPlan Hash {fp.floorplan_id}."
                                        )
                                    )

                    except FloorPlanAnalysisResult.DoesNotExist:
                        self.stderr.write(
                            self.style.WARNING(
                                f"  No FloorPlanAnalysisResult found for User '{user_id_csv}', Property '{property_id_csv}'."
                            )
                        )
                    except FloorPlanAnalysisResult.MultipleObjectsReturned:
                        self.stderr.write(
                            self.style.ERROR(
                                f"  Multiple FloorPlanAnalysisResults found for User '{user_id_csv}', Property '{property_id_csv}'. Skipping."
                            )
                        )
                    except Exception as e:
                        self.stderr.write(
                            self.style.ERROR(
                                f"  Error querying data for {property_key}: {e}"
                            )
                        )

                    # Add row to report data, joining lists with commas
                    report_data.append(
                        [
                            property_id_csv,
                            user_id_csv,
                            property_url_csv,
                            original_fp_urls_csv,
                            ", ".join(processed_fp_hashes),
                            ", ".join(all_floors_urls),
                            ", ".join(total_area_urls),
                        ]
                    )

        except FileNotFoundError:
            raise CommandError(f"Input CSV file not found: {input_csv_path}")
        except Exception as e:
            logger.exception(
                f"An unexpected error occurred reading input CSV or querying data: {e}"
            )
            raise CommandError(f"An unexpected error occurred: {e}")

        # Write the final report CSV
        try:
            with open(
                output_report_csv_path, mode="w", newline="", encoding="utf-8"
            ) as outfile:
                writer = csv.writer(outfile)
                writer.writerows(report_data)
            self.stdout.write(
                self.style.SUCCESS(
                    f"Successfully generated analysis report: {output_report_csv_path}"
                )
            )
        except IOError as e:
            raise CommandError(
                f"Error writing output report CSV file {output_report_csv_path}: {e}"
            )

class Command(BaseCommand):
    help = """
    Utilities for historical floorplan data:
    1. (--backfill-raw): Backfills AllFloorsCsvRawRow from existing structured data
       for existing FloorPlan records and triggers backup. Uses current data.
    2. (--reanalyze): Triggers re-analysis task for EXISTING FloorPlan records based on
       their stored original_url. NOTE: Webhook processing uses URL hash ID and will
       UPDATE the corresponding FloorPlan record.
    3. (--find-missing): Finds FloorPlanAnalysisResults WITHOUT linked FloorPlan records
       and triggers analysis for them (attempts to create missing FloorPlans).
       Requires logic to find the original URL.

    USE WITH CAUTION - Re-analysis can cause high load. Test with limits/filters.
    """

    def add_arguments(self, parser):
        parser.add_argument(
            "--backfill-raw",
            action="store_true",
            help="Backfill AllFloorsCsvRawRow for existing FloorPlans and trigger backup.",
        )
        parser.add_argument(
            "--reanalyze",
            action="store_true",
            help="Trigger re-analysis task for EXISTING FloorPlan records (should update).",
        )
        parser.add_argument(
            "--find-missing",
            action="store_true",
            help="Find AnalysisResults without FloorPlans and trigger analysis for them.",
        )
        parser.add_argument(
            "--floorplan-id",
            type=str,
            help="(For --backfill-raw / --reanalyze) Process only a specific existing FloorPlan Hash ID.",
        )
        parser.add_argument(
            "--user-id",
            type=str,
            help="Filter by User ID (applies to all modes).",
        )
        parser.add_argument(
            "--property-id",
            type=str,
            help="Filter by Property ID (requires --user-id, applies to all modes).",
        )
        parser.add_argument(
            "--analysis-result-id",
            type=int,
            help="(For --find-missing) Process only a specific FloorPlanAnalysisResult PK.",
        )
        parser.add_argument(
            "--limit",
            type=int,
            help="Limit the number of items (FloorPlans or AnalysisResults) to process.",
        )
        parser.add_argument(
            "--delay",
            type=float,
            default=2.0,
            help="Delay in seconds between triggering re-analysis tasks (default: 2.0).",
        )
        parser.add_argument(
            "--start-index",
            type=int,
            default=0,
            help="Start processing from this index (0-based) in the queryset (useful for resuming).",
        )
        parser.add_argument(
            "--batch-size",
            type=int,
            default=100,
            help="Log progress message every N items processed.",
        )

    def handle(self, *args, **options):
        backfill = options["backfill_raw"]
        reanalyze = options["reanalyze"]
        find_missing = options["find_missing"]
        fp_id_filter = options["floorplan_id"]
        user_id_filter = options["user_id"]
        prop_id_filter = options["property_id"]
        ar_id_filter = options["analysis_result_id"]
        limit = options["limit"]
        delay = options["delay"]
        start_index = options["start_index"]
        batch_size = options["batch_size"]

        # --- Argument Validation ---
        if not backfill and not reanalyze and not find_missing:
            raise CommandError(
                "Specify at least one action: --backfill-raw, --reanalyze, or --find-missing."
            )
        if prop_id_filter and not user_id_filter:
            raise CommandError("--property-id requires --user-id.")
        if fp_id_filter and find_missing:
            raise CommandError("--floorplan-id cannot be used with --find-missing.")
        if ar_id_filter and (backfill or reanalyze):
            raise CommandError(
                "--analysis-result-id can only be used with --find-missing."
            )

        # --- Initialize Counters ---
        processed_count_fp = 0
        processed_count_missing = 0
        errors_backfill = 0
        errors_backup = 0
        errors_reanalyze_existing = 0
        errors_reanalyze_missing = 0

        # --- Mode 1 & 2: Process EXISTING FloorPlans ---
        if backfill or reanalyze:
            self.stdout.write(
                self.style.MIGRATE_HEADING(
                    "Processing Existing FloorPlans (--backfill-raw / --reanalyze)"
                )
            )
            queryset_fp = self._get_filtered_floorplan_queryset(
                fp_id_filter, user_id_filter, prop_id_filter
            )
            total_fp_count = queryset_fp.count()
            self.stdout.write(
                f"Found {total_fp_count} existing FloorPlan(s) matching criteria."
            )

            effective_fp_count = self._apply_limit_offset(
                total_fp_count, limit, start_index
            )
            if effective_fp_count <= 0:
                self.stdout.write(
                    "No existing floorplans to process with current limit/start_index."
                )
            else:
                self.stdout.write(
                    f"Processing {effective_fp_count} FloorPlan(s) starting from index {start_index}..."
                )
                # Apply slicing only before iteration
                queryset_fp_limited = queryset_fp[
                    start_index : start_index + effective_fp_count
                ]

                for i, floor_plan in enumerate(queryset_fp_limited, start=start_index):
                    current_index = i
                    processed_count_fp += 1
                    self.stdout.write(
                        f"\n--- Processing FP {processed_count_fp}/{effective_fp_count} (Index:{current_index}, Hash:{floor_plan.floorplan_id}, PK:{floor_plan.pk}) ---"
                    )

                    if not self._check_analysis_result(floor_plan):
                        continue

                    # -- Backfill --
                    if backfill:
                        errors_b, errors_bk = self._run_backfill_and_backup(floor_plan)
                        errors_backfill += errors_b
                        errors_backup += errors_bk

                    # -- Reanalyze Existing --
                    if reanalyze:
                        errors_reanalyze_existing += self._run_reanalyze_existing(
                            floor_plan, delay
                        )

                    if processed_count_fp > 0 and processed_count_fp % batch_size == 0:
                        self.stdout.write(
                            f"--- Processed batch ending at index {current_index} ---"
                        )

        # --- Mode 3: Find Missing FloorPlans and Reanalyze ---
        if find_missing:
            self.stdout.write(
                self.style.MIGRATE_HEADING(
                    "\nFinding AnalysisResults Missing FloorPlans (--find-missing)"
                )
            )
            queryset_missing = self._get_filtered_missing_queryset(
                ar_id_filter, user_id_filter, prop_id_filter
            )
            total_missing_count = queryset_missing.count()
            self.stdout.write(
                f"Found {total_missing_count} AnalysisResult(s) missing FloorPlans matching criteria."
            )

            effective_missing_count = self._apply_limit_offset(
                total_missing_count, limit, start_index
            )
            if effective_missing_count <= 0:
                self.stdout.write(
                    "No missing floorplans to process with current limit/start_index."
                )
            else:
                self.stdout.write(
                    f"Processing {effective_missing_count} AnalysisResult(s) starting from index {start_index}..."
                )
                # Apply slicing only before iteration
                queryset_missing_limited = queryset_missing[
                    start_index : start_index + effective_missing_count
                ]

                for i, analysis_result in enumerate(
                    queryset_missing_limited, start=start_index
                ):
                    current_index = i
                    processed_count_missing += 1
                    self.stdout.write(
                        f"\n--- Processing Missing FP {processed_count_missing}/{effective_missing_count} (Index:{current_index}, AR PK:{analysis_result.pk}) ---"
                    )

                    errors_reanalyze_missing += self._run_reanalyze_missing(
                        analysis_result, delay
                    )

                    if (
                        processed_count_missing > 0
                        and processed_count_missing % batch_size == 0
                    ):
                        self.stdout.write(
                            f"--- Processed batch ending at index {current_index} ---"
                        )

        # --- Final Summary ---
        self._print_summary(
            backfill,
            reanalyze,
            find_missing,
            processed_count_fp,
            processed_count_missing,
            errors_backfill,
            errors_backup,
            errors_reanalyze_existing,
            errors_reanalyze_missing,
        )

    # --- Helper Methods for Querying and Actions ---

    def _get_filtered_floorplan_queryset(
        self, fp_id_filter, user_id_filter, prop_id_filter
    ):
        """Builds and filters the queryset for existing FloorPlans."""
        queryset = (
            FloorPlan.objects.select_related("analysis_result", "all_floors_data")
            .prefetch_related(  # Keep prefetches needed for backfill
                "all_floors_data__csv_floors__rooms__pixel_data",
                "all_floors_data__csv_floors__rooms__dimensions",
                "all_floors_data__csv_floors__rooms__scaling_factors",
            )
            .order_by("id")
        )

        if fp_id_filter:
            self.stdout.write(f"Filtering for FloorPlan Hash ID: {fp_id_filter}")
            queryset = queryset.filter(floorplan_id=fp_id_filter)
        elif user_id_filter:
            self.stdout.write(f"Filtering for User ID: {user_id_filter}")
            queryset = queryset.filter(analysis_result__user_id=user_id_filter)
            if prop_id_filter:
                self.stdout.write(f"Filtering for Property ID: {prop_id_filter}")
                queryset = queryset.filter(analysis_result__property_id=prop_id_filter)
        return queryset

    def _get_filtered_missing_queryset(
        self, ar_id_filter, user_id_filter, prop_id_filter
    ):
        """Builds and filters the queryset for missing FloorPlans (via AnalysisResult)."""
        results_with_fp = FloorPlan.objects.values_list(
            "analysis_result_id", flat=True
        ).distinct()
        queryset = FloorPlanAnalysisResult.objects.exclude(
            id__in=results_with_fp
        ).order_by("id")

        if ar_id_filter:
            self.stdout.write(f"Filtering for AnalysisResult PK: {ar_id_filter}")
            queryset = queryset.filter(pk=ar_id_filter)
        elif user_id_filter:

    def _apply_limit_offset(self, total_count, limit, start_index):
        """Helper to calculate effective count after limit/offset."""
        if start_index >= total_count:
            return 0
        effective_count = total_count - start_index
        if limit is not None:
            effective_count = min(limit, effective_count)
        return effective_count if effective_count > 0 else 0

            help="(Optional) Filter report for a specific User ID.",

            type=str,
            help="(Optional) Filter report for a specific Property ID (requires --user-id).",

        parser.add_argument(

            type=str,
            help="(Optional) Filter report for a specific FloorPlan Hash ID (requires --user-id and --property-id).",
        )
        # Output file remains positional and required
        parser.add_argument(
            "output_file", type=str, help="Path to the output CSV file."
        )

    def handle(self, *args, **options):
        user_id_filter = options["user_id"]
        property_id_filter = options["property_id"]
        floorplan_id_filter = options["floorplan_id"]
        output_file = options["output_file"]

        # --- Input Validation ---
        if property_id_filter and not user_id_filter:
            raise CommandError("--property-id requires --user-id to be specified.")
        if floorplan_id_filter and not (user_id_filter and property_id_filter):
            raise CommandError(
                "--floorplan-id requires both --user-id and --property-id."
            )

        # --- Determine Query Scope ---
        all_floors_data_ids = None
        report_scope = "GLOBAL"

        if user_id_filter and property_id_filter:
            report_scope = f"User: {user_id_filter}, Property: {property_id_filter}"
            try:
                # Find the single AnalysisResult matching the user/property
                analysis_result = FloorPlanAnalysisResult.objects.get(
                    user_id=user_id_filter, property_id=property_id_filter
                )
                self.stdout.write(
                    f"Found AnalysisResult PK: {analysis_result.pk} for specified filters."
                )

                # Find FloorPlan(s) linked to this AnalysisResult
                floorplans_qs = FloorPlan.objects.filter(
                    analysis_result=analysis_result
                )
                if floorplan_id_filter:
                    report_scope += f", FloorPlan: {floorplan_id_filter}"
                    floorplans_qs = floorplans_qs.filter(
                        floorplan_id=floorplan_id_filter
                    )
                    self.stdout.write(
                        f"Filtering for FloorPlan Hash ID: {floorplan_id_filter}"
                    )

                floorplan_pks = list(floorplans_qs.values_list("pk", flat=True))
                if not floorplan_pks:
                    self.stdout.write(
                        self.style.WARNING(
                            f"No FloorPlan records found matching the specified criteria. Report may be incomplete."
                        )

                else:
                    self.stdout.write(
                        f"Found {len(floorplan_pks)} FloorPlan(s) for this analysis."
                    )
                    # Get AllFloorsData IDs associated with the found FloorPlan(s)

                        AllFloorsData.objects.filter(

                raise CommandError(
                    f"No FloorPlanAnalysisResult found for User '{user_id_filter}', Property '{property_id_filter}'."
                )
            except FloorPlanAnalysisResult.MultipleObjectsReturned:
                raise CommandError(
                    f"Multiple FloorPlanAnalysisResult found for User '{user_id_filter}', Property '{property_id_filter}'. Cannot generate specific report."
                )
        else:
            self.stdout.write(
                "No user/property filters provided. Generating GLOBAL report."
            )
            # For global report, all_floors_data_ids remains None, queries will use .all()

            "max_area_metric",  # Col 19
            "max_area_imperial",  # Col 20
            "max_area_pixels",  # Col 21
            "actual_area_pixels",  # Col 22
            "pixel_ratio",  # Col 23
            "scale_metric",  # Col 24
            "scale_imperial",  # Col 25
            "calculated_sq_area_metric",  # Col 26
            "calculated_floor_total_sq_area_metric",  # Col 27

            "calculated_floor_total_sq_area_imperial",  # Col 29

import csv
import logging
import time

from django.conf import settings
from django.core.management.base import BaseCommand, CommandError

from floorplan.tasks import process_floorplan_analysis

logger = logging.getLogger(__name__)


class Command(BaseCommand):
    help = "Reads a CSV with property and floorplan URLs, triggers analysis task for each property with a delay."

    def add_arguments(self, parser):
        parser.add_argument(
            "input_csv_path", type=str, help="Path to the input CSV file."
        )
        parser.add_argument(
            "output_log_csv_path",
            type=str,
            help="Path to the output CSV file logging triggered tasks.",
        )
        parser.add_argument(
            "--delay",
            type=int,
            default=60,
            help="Delay in seconds between triggering analysis for each property.",
        )
        parser.add_argument(
            "--start-row",
            type=int,
            default=1,
            help="1-based row number to start processing from (including header row).",
        )
        parser.add_argument(
            "--limit",
            type=int,
            default=None,
            help="Maximum number of properties (rows) to process.",
        )

    def handle(self, *args, **options):
        input_csv_path = options["input_csv_path"]
        output_log_csv_path = options["output_log_csv_path"]
        delay_seconds = options["delay"]
        start_row = options["start_row"]
        limit = options["limit"]

        processed_count = 0
        triggered_rows_log = []

        self.stdout.write(f"Reading input CSV: {input_csv_path}")
        # ... (rest of setup logging) ...

        try:
            with open(input_csv_path, mode="r", encoding="utf-8-sig") as infile:
                reader = csv.DictReader(infile)
                headers = reader.fieldnames

                if not all(
                    h in headers
                    for h in [
                        "Property ID",
                        "User ID",
                        "Property URL",
                        "Floorplan URLs",
                    ]
                ):
                    raise CommandError(
                        "Input CSV must contain columns: 'Property ID', 'User ID', 'Property URL', 'Floorplan URLs'"
                    )

                triggered_rows_log.append(headers + ["Status", "Timestamp"])

                for i, row in enumerate(reader, start=2):
                    if i < start_row:
                        continue
                    if limit is not None and processed_count >= limit:
                        self.stdout.write(
                            f"Reached processing limit of {limit}. Stopping."
                        )
                        break

                    property_id_csv = row.get("Property ID", "").strip()
                    user_id_csv = row.get("User ID", "").strip()
                    property_url_csv = row.get("Property URL", "").strip()
                    floorplan_urls_str = row.get("Floorplan URLs", "").strip()

                    if not property_id_csv or not user_id_csv or not floorplan_urls_str:
                        # ... (skip logging) ...
                        continue

                    floorplan_urls = [
                        url.strip()
                        for url in floorplan_urls_str.split(",")
                        if url.strip()
                    ]

                    if not floorplan_urls:
                        # ... (skip logging) ...
                        continue

                    # --- Build the payload with numerical index keys ---
                    floorplans_payload = {}
                    for idx, fp_url in enumerate(floorplan_urls):
                        # Use the string representation of the index as the key
                        floorplans_payload[str(idx)] = {  # <--- CHANGE HERE
                            "url": fp_url,
                            "notes": "",
                        }
                    # --- End payload modification ---

                    # Trigger the Celery task
                    try:
                        self.stdout.write(
                            f"Processing row {i}: Triggering analysis for Property ID '{property_id_csv}', User ID '{user_id_csv}' with {len(floorplans_payload)} floorplan(s)..."
                        )
                        process_floorplan_analysis.delay(
                            user_id=str(user_id_csv),
                            property_id=str(property_id_csv),
                            floorplans=floorplans_payload,  # Pass the dict with '0', '1', ... keys
                        )
                        # ... (rest of try block: success logging, delay) ...
                        status_msg = "Triggered"
                        processed_count += 1
                        if delay_seconds > 0:
                            self.stdout.write(
                                f"  Waiting for {delay_seconds} seconds..."
                            )
                            time.sleep(delay_seconds)

                    except Exception as e:
                        # ... (error logging) ...
                        status_msg = f"Error - {e}"

                    # Log the outcome
                    log_row = [row.get(h, "") for h in headers] + [
                        status_msg,
                        time.strftime("%Y-%m-%d %H:%M:%S"),
                    ]
                    triggered_rows_log.append(log_row)

        except FileNotFoundError:
            raise CommandError(f"Input CSV file not found: {input_csv_path}")
        except Exception as e:
            logger.exception(f"An unexpected error occurred during CSV processing: {e}")
            raise CommandError(f"An unexpected error occurred: {e}")

        # Write the output log CSV
        # ... (rest of file: writing log csv) ...
        try:
            with open(
                output_log_csv_path, mode="w", newline="", encoding="utf-8"
            ) as outfile:
                writer = csv.writer(outfile)
                writer.writerows(triggered_rows_log)
            self.stdout.write(
                self.style.SUCCESS(
                    f"Finished processing. Triggered {processed_count} tasks. Log saved to {output_log_csv_path}"
                )
            )
        except IOError as e:
            raise CommandError(
                f"Error writing output log CSV file {output_log_csv_path}: {e}"
            )

import requests
from django.conf import settings
from django.forms.models import model_to_dict

from floorplan_service.config.logging_config import configure_logger

logger = configure_logger(__name__)


def backup_floorplan(instance):
    try:
        logger.info(
            "Starting backup process for FloorPlan ID: %s", instance.floorplan_id
        )
        # Build the analysis result payload (exclude auto fields)
        analysis_result_payload = {
            "message": instance.analysis_result.message,
            "user_id": instance.analysis_result.user_id,
            "property_id": instance.analysis_result.property_id,
        }

        # Build the main floorplan payload – only required fields.
        payload = {
            "analysis_result": analysis_result_payload,
            "floorplan_id": instance.floorplan_id,
            "original_url": instance.original_url,
        }

        # Build the AllFloorsData payload if available.
        if hasattr(instance, "all_floors_data") and instance.all_floors_data:
            all_floors = instance.all_floors_data
            all_floors_payload = {
                "json_file_url": all_floors.json_file_url,
                "csv_url": all_floors.csv_url,
                "total_area_csv_url": all_floors.total_area_csv_url,
                "image_labelme_side_by_side_url": all_floors.image_labelme_side_by_side_url,
                "notes": all_floors.notes,
                # Initialize lists for nested structures
                "backup_csv_floors": [],
                "backup_all_floors_raw_rows": [],  # <-- Initialize raw rows list
                "backup_total_area_data": [],  # <-- Initialize total area list (if backing up)
            }

            # Process Structured CSV floors (CsvFloor, CsvRoom, etc.)
            csv_floors_list = []
            for csv_floor in all_floors.csv_floors.all():
                csv_floor_payload = {
                    "floor_name": csv_floor.floor_name,
                    "calculated_total_area_metric": csv_floor.calculated_total_area_metric,
                    "calculated_total_area_imperial": csv_floor.calculated_total_area_imperial,
                }
                rooms_list = []
                for room in csv_floor.rooms.all():
                    room_payload = {
                        "room_name": room.room_name,
                        "is_segment": room.is_segment,
                        "room_id": room.room_id,
                        "no_of_doors": room.no_of_doors,
                        "no_of_windows": room.no_of_windows,
                        "no_of_room_points": room.no_of_room_points,
                    }
                    # Include pixel data if available
                    if hasattr(room, "pixel_data"):
                        pd = room.pixel_data
                        room_payload["backup_pixel_data"] = {
                            "min_x_pixels": pd.min_x_pixels,
                            "min_y_pixels": pd.min_y_pixels,
                            "max_x_pixels": pd.max_x_pixels,
                            "max_y_pixels": pd.max_y_pixels,
                            "max_area_pixels": pd.max_area_pixels,
                            "actual_area_pixels": pd.actual_area_pixels,
                            "pixel_ratio": pd.pixel_ratio,
                        }
                    # Include dimensions if available
                    if hasattr(room, "dimensions"):
                        dims = room.dimensions
                        room_payload["backup_dimensions"] = {
                            "dimensions_imperial": dims.dimensions_imperial,
                            "dimensions_metric": dims.dimensions_metric,
                            "max_area_metric": dims.max_area_metric,
                            "max_area_imperial": dims.max_area_imperial,
                            "calculated_sq_area_metric": dims.calculated_sq_area_metric,
                            "calculated_area_imperial": dims.calculated_area_imperial,
                        }
                    # Include scaling factors if available
                    if hasattr(room, "scaling_factors"):
                        sf = room.scaling_factors
                        room_payload["backup_scaling_factors"] = {
                            "scale_metric": sf.scale_metric,
                            "scale_imperial": sf.scale_imperial,
                        }
                    rooms_list.append(room_payload)
                csv_floor_payload["backup_rooms"] = rooms_list
                csv_floors_list.append(csv_floor_payload)
            all_floors_payload["backup_csv_floors"] = csv_floors_list
            payload["backup_all_floors_data"] = all_floors_payload

            # --- Process Raw Row Data (AllFloorsCsvRawRow) ---
            raw_rows_list = []
            # Efficiently serialize using model_to_dict or values()
            raw_row_queryset = all_floors.all_floors_raw_rows.all().order_by(
                "id"
            )  # Order for consistency
            for raw_row in raw_row_queryset:
                # Exclude the foreign key object itself to avoid circular refs
                raw_row_dict = model_to_dict(raw_row, exclude=["all_floors_data"])
                # Could add the parent ID explicitly if needed
                # raw_row_dict['all_floors_data_id'] = raw_row.all_floors_data_id
                raw_rows_list.append(raw_row_dict)
            all_floors_payload["backup_all_floors_raw_rows"] = raw_rows_list
            logger.debug("Serialized %d raw CSV rows.", len(raw_rows_list))
            # --- END Raw Row Data ---

            # --- Process Total Area Data (TotalAreaData) --- Optional, include if needed
            logger.debug("Serializing total area data for backup...")
            total_area_list = []
            total_area_queryset = all_floors.total_area_data.all().order_by(
                "id"
            )  # Order for consistency
            for ta_data in total_area_queryset:
                # Exclude the foreign key object
                ta_data_dict = model_to_dict(ta_data, exclude=["all_floors_data"])
                total_area_list.append(ta_data_dict)
            all_floors_payload["backup_total_area_data"] = total_area_list
            logger.debug("Serialized %d total area data rows.", len(total_area_list))
            # --- END Total Area Data ---

            payload["backup_all_floors_data"] = all_floors_payload
        else:
            logger.info(
                "No AllFloorsData found for FloorPlan ID %s, skipping related backup data.",
                instance.floorplan_id,
            )

        # Process PlanFloor data.
        logger.debug("Serializing PlanFloor data...")
        backup_plan_floors = []
        for pf in instance.plan_floors.all():
            pf_payload = {
                "floor": pf.floor,
                "label_me_url": pf.label_me_url,
                "json_file_url": pf.json_file_url,
                "image_url": pf.image_url,
                "labelme_image_url": pf.labelme_image_url,
                "csv_url": pf.csv_url,
                "image_side_by_side_url": pf.image_side_by_side_url,
            }
            backup_plan_floors.append(pf_payload)
        payload["backup_plan_floors"] = backup_plan_floors
        logger.debug("Serialized %d PlanFloor entries.", len(backup_plan_floors))

        # Send the payload to the backup endpoint.
        backup_url = f"{settings.BACKUP_SERVICE_URL}/api/backup/floorplan/"
        logger.info(
            "Sending backup payload for FloorPlan ID %s to %s",
            instance.floorplan_id,
            backup_url,
        )
        response = requests.post(backup_url, json=payload, timeout=5)
        # If there is a 400 or any HTTP error, log the response details.
        response.raise_for_status()
        logger.info(
            "Backup successful for FloorPlan ID %s. Response status: %s",
            instance.floorplan_id,
            response.status_code,
        )
    except requests.exceptions.Timeout:
        logger.error(
            "Timeout occurred while sending backup for FloorPlan ID %s to %s",
            instance.floorplan_id,
            backup_url,
        )
        # Re-raise or handle as needed - maybe retry?
        raise Exception(
            f"Timeout backing up floorplan {instance.floorplan_id}"
        ) from None
    except requests.HTTPError as http_err:
        # Log response content to help pinpoint validation errors on the backup service
        error_details = "No response body"
        try:
            error_details = (
                http_err.response.text
            )  # Use .text for JSON or other text formats
        except Exception:
            logger.warning("Could not decode error response body for backup failure.")

        logger.error(
            "HTTP Error backing up FloorPlan ID %s: %s - Status Code: %s - Details: %s",
            instance.floorplan_id,
            str(http_err),
            http_err.response.status_code,
            error_details,
        )
        raise Exception(
            f"HTTP error backing up floorplan {instance.floorplan_id}: {str(http_err)}"
        ) from http_err
    except requests.RequestException as req_err:
        logger.error(
            "Request Exception during backup for FloorPlan ID %s: %s",
            instance.floorplan_id,
            str(req_err),
        )
        raise Exception(
            f"Request error backing up floorplan {instance.floorplan_id}: {str(req_err)}"
        ) from req_err
    except Exception as e:
        # Catch any other unexpected errors during serialization or processing
        logger.exception(
            "Unexpected error during backup process for FloorPlan ID %s: %s",
            instance.floorplan_id,
            str(e),
        )
        raise Exception(
            f"Failed to complete backup for floorplan {instance.floorplan_id}"
        ) from e
